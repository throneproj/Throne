cmake_minimum_required(VERSION 3.20)
project(ThroneMacSystemProxyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
get_filename_component(THRONE_ROOT ../.. ABSOLUTE)
add_executable(macos-system-proxy-tests main.cpp ${THRONE_ROOT}/src/sys/macos/ProxyJournal.cpp)
target_include_directories(macos-system-proxy-tests PRIVATE ${THRONE_ROOT})
target_link_libraries(macos-system-proxy-tests PRIVATE Qt6::Core)
if(APPLE)
    # Compile/link the actual platform adapter too; tests never invoke it or alter host settings.
    target_sources(macos-system-proxy-tests PRIVATE ${THRONE_ROOT}/src/sys/macos/SystemProxy.cpp)
    target_link_libraries(macos-system-proxy-tests PRIVATE "-framework SystemConfiguration" "-framework Security" "-framework CoreFoundation")
endif()
enable_testing()
add_test(NAME macos-system-proxy-regressions COMMAND macos-system-proxy-tests)
